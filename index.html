// RENDERIZAR PAINEL DO DESPACHANTE function renderDashboard() { const tbody = document.getElementById('clients-table-body'); if (!tbody) return; tbody.innerHTML = ''; let total = clientsData.length; let pendentes = 0; let prontos = 0; let deferidos = 0; const searchEl = document.getElementById('search-input'); const statusEl = document.getElementById('status-filter'); const filterText = searchEl ? searchEl.value.toLowerCase() : ''; const filterStatus = statusEl ? statusEl.value : 'ALL'; clientsData.forEach(client => { // Métricas const docsCount = Object.values(client.docs).filter(d => d.uploaded).length; if (client.status === 'Deferido') deferidos++; else if (docsCount === REQUIRED_DOCUMENTS.length || client.status === 'Pronto') prontos++; else pendentes++; // Filtro const matchesSearch = (client.fullname || '').toLowerCase().includes(filterText) || (client.cpf || '').includes(filterText); let matchesStatus = true; if (filterStatus === 'Pendente') matchesStatus = docsCount < REQUIRED_DOCUMENTS.length; else if (filterStatus === 'Pronto') matchesStatus = client.status === 'Pronto' || docsCount === REQUIRED_DOCUMENTS.length; else if (filterStatus !== 'ALL') matchesStatus = client.status === filterStatus; if (!matchesSearch || !matchesStatus) return; // Status Badge CSS let badgeClass = 'bg-amber-100 text-amber-800 border-amber-200'; if (client.status === 'Pronto' || docsCount === REQUIRED_DOCUMENTS.length) { badgeClass = 'bg-blue-100 text-blue-800 border-blue-200'; } else if (client.status === 'Deferido') { badgeClass = 'bg-emerald-100 text-emerald-800 border-emerald-200'; } const tr = document.createElement('tr'); tr.className = "hover:bg-slate-50 transition-colors border-b border-slate-100"; tr.innerHTML = `
${client.fullname || 'Sem nome'}
ID: #${client.id}
${client.cpf || '---'}
${client.phone || 'Sem telefone'}
${docsCount}/${REQUIRED_DOCUMENTS.length}
${docsCount === REQUIRED_DOCUMENTS.length && client.status !== 'Deferido' ? 'Pronto p/ Protocolo' : client.status} `; tbody.appendChild(tr); }); // Atualizar cards de estatísticas const totalEl = document.getElementById('stat-total'); const pendenteEl = document.getElementById('stat-pendente'); const prontoEl = document.getElementById('stat-pronto'); const deferidoEl = document.getElementById('stat-deferido'); if (totalEl) totalEl.innerText = total; if (pendenteEl) pendenteEl.innerText = pendentes; if (prontoEl) prontoEl.innerText = prontos; if (deferidoEl) deferidoEl.innerText = deferidos; try { if (window.lucide) lucide.createIcons(); } catch(e) {} } // CARREGAR DADOS DO CLIENTE NO PORTAL DO CLIENTE // LEITURA RÍGIDA DE URL E HASH PARA MODO CLIENTE let isClientModeOnly = false; function checkUrlMode() { loadFromLocalStorage(); const hash = window.location.hash || ''; const searchParams = new URLSearchParams(window.location.search); const href = window.location.href; const isNovoCadastro = hash.includes('cadastro') || searchParams.get('novo-cadastro') === 'true' || href.includes('novo-cadastro=true'); const clientIdParam = searchParams.get('client') || (hash.includes('client=') ? hash.split('client=')[1] : null); if (isNovoCadastro || clientIdParam) { isClientModeOnly = true; document.documentElement.classList.add('mode-cliente-only'); const mainHeader = document.getElementById('main-header'); const clientHeader = document.getElementById('client-only-header'); const btnBack = document.getElementById('btn-back-despachante'); if (mainHeader) mainHeader.classList.add('hidden'); if (clientHeader) clientHeader.classList.remove('hidden'); if (btnBack) btnBack.classList.add('hidden'); if (isNovoCadastro) { prepareBlankRegistration(); } else if (clientIdParam) { loadClientIntoForm(clientIdParam); } switchView('cliente'); } else { isClientModeOnly = false; document.documentElement.classList.remove('mode-cliente-only'); const mainHeader = document.getElementById('main-header'); const clientHeader = document.getElementById('client-only-header'); const btnBack = document.getElementById('btn-back-despachante'); if (mainHeader) mainHeader.classList.remove('hidden'); if (clientHeader) clientHeader.classList.add('hidden'); if (btnBack) btnBack.classList.remove('hidden'); renderDashboard(); if (clientsData.length > 0) loadClientIntoForm(clientsData[0].id); switchView('despachante'); } } // PREPARAR CADASTRO LIMPO DO ZERO
